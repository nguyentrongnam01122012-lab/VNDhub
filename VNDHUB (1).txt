-- ====== GENERATE KEY ======
local function generateUserKey()
    local period = math.floor(os.time() / 3600 / PERIOD_HOURS)
    local seed   = (period * 9999 + tonumber(userId)) % INT32_MAX
    math.randomseed(seed)
    local s = ""
    for _ = 1, KEY_LENGTH do
        s = s .. tostring(math.random(0, 9))
    end
    return KEY_PREFIX .. s
end

-- ====== VALIDATE FORMAT KEY ======
local function isValidKeyFormat(k)
    if type(k) ~= "string" then return false end
    return k:match("^VND_[%w_]+$") ~= nil
end

end

-- ====== PARSE EXPIRE ======
local function parseExpireAt(val)
    if val == nil then return 0 end
    local n = tonumber(val)
    if n then return n end
    local s = tostring(val)
    local y, m, d = s:match("^(%d%d%d%d)-(%d%d)-(%d%d)")
    if y then
        return os.time({year=tonumber(y),month=tonumber(m),day=tonumber(d),hour=23,min=59,sec=59})
    end
    local d2, m2, y2 = s:match("^(%d%d)/(%d%d)/(%d%d%d%d)")
    if d2 then
        return os.time({year=tonumber(y2),month=tonumber(m2),day=tonumber(d2),hour=23,min=59,sec=59})
    end
    return 0
end

-- ====== FIX: Xử lý Firebase trả về array node ======
local function resolveFirebaseData(data)
    if data.key ~= nil or data.expireAt ~= nil or data.expire ~= nil or data.type ~= nil then
        return data
    end
    local latest = nil
    local latestExp = -1
    for _, v in pairs(data) do
        if type(v) == "table" then
            local exp = parseExpireAt(v.expireAt or v.expire)
            if exp > latestExp then
                latestExp = exp
                latest = v
            end
        end
    end
    return latest
end

-- ====== FIREBASE ======
local function saveKeyToFirebase(key)
    local keyId      = key:gsub("[^%w_]", "_")
    local url        = DB .. "/keys/" .. keyId .. ".json"
    local expireTime = os.time() + KEY_DURATION_SEC

    local data = HttpService:JSONEncode({
        key      = key,
        userId   = userId,
        expireAt = expireTime,
        type     = "normal",
        created  = os.date("%d/%m/%Y %H:%M")
    })

    local ok, err = pcall(function()
        HttpService:RequestAsync({
            Url     = url,
            Method  = "PUT",
            Headers = { ["Content-Type"] = "application/json" },
            Body    = data
        })
    end)
    if not ok then warn("[VND] Firebase lỗi: " .. tostring(err)) end
    return ok
end

local function updateKeyWithUserId(key)
    local keyId = key:gsub("[^%w_]", "_")
    local url   = DB .. "/keys/" .. keyId .. ".json"

    local data = HttpService:JSONEncode({
        userId = userId
    })

    local ok, err = pcall(function()
        HttpService:RequestAsync({
            Url     = url,
            Method  = "PATCH",
            Headers = { ["Content-Type"] = "application/json" },
            Body    = data
        })
    end)
    if not ok then warn("[VND] Firebase update lỗi: " .. tostring(err)) end
    return ok
end

local function make(cls, props, parent)
    local o = Instance.new(cls)
    for k, v in pairs(props) do o[k] = v end
    if parent then o.Parent = parent end
    return o
end

    Instance.new("UICorner", p).CornerRadius = UDim.new(0, r)

local function grad(c1, c2, rot, p)
    local g = Instance.new("UIGradient", p)

    g.Rotation = rot

        showStatus("Lỗi kết nối Firebase!", Color3.fromRGB(255,100,0), "⚠️")
        task.wait(COOLDOWN_SEC); StatusBox.Visible = false; checking = false; setButtonsEnabled(true)

    local dok, raw = pcall(function() return HttpService:JSONDecode(res) end)
    if not dok or raw == nil then
        showStatus("Dữ liệu lỗi! Thử lại.", Color3.fromRGB(255,100,0), "⚠️")
        task.wait(COOLDOWN_SEC); StatusBox.Visible = false; checking = false; setButtonsEnabled(true)

        task.wait(COOLDOWN_SEC); StatusBox.Visible = false; checking = false; setButtonsEnabled(true)
        return

    local isKeyEmpty = (data.userId == nil or data.userId == "")

local function getTargetData(model)

    if not model or not model.Parent then
        return nil, nil
    end

    local hum = model:FindFirstChildOfClass("Humanoid")

    if not hum or hum.Health <= 0 then
        return nil, nil
    end

    local root = model:FindFirstChild("HumanoidRootPart")

    if not root then
        return nil, nil
    end

    return hum, root
end

--========================================================
-- CACHE TARGET
--========================================================

local function tryAddTarget(obj)

    if not obj:IsA("Model") then
        return
    end

    if obj.Name ~= _G.Settings.TargetName then
        return
    end

    if targets[obj] then
        return
    end

    local hum, root = getTargetData(obj)

    if hum and root then
        targets[obj] = true
    end
end

--========================================================
-- REMOVE TARGET
--========================================================

local function removeTarget(obj)

    targets[obj] = nil

    if lockedTarget == obj then
        lockedTarget = nil
        lockedPosition = nil
    end
end

--========================================================
-- INITIAL CACHE
--
-- GetDescendants chỉ chạy 1 lần khi script khởi động.
--========================================================

for _, obj in ipairs(Workspace:GetDescendants()) do
    tryAddTarget(obj)
end

--========================================================
-- NEW OBJECT
--========================================================

Workspace.DescendantAdded:Connect(function(obj)

    if obj.Name ~= _G.Settings.TargetName then
        return
    end

    task.defer(function()

        if obj.Parent then
            tryAddTarget(obj)
        end

    end)

end)

--========================================================
-- REMOVED OBJECT
--========================================================

Workspace.DescendantRemoving:Connect(function(obj)

    if targets[obj] then
        removeTarget(obj)
    end

end)

--========================================================
-- FIND TARGET FROM CACHE
--========================================================

local function findTarget()

    for target in pairs(targets) do

        local hum, root =
            getTargetData(target)

        if hum and root then
            return target, hum, root
        end

        targets[target] = nil

    end

    return nil, nil, nil
end

--========================================================
-- CLEAR
--========================================================

local function clearTarget()

    lockedTarget = nil
    lockedPosition = nil

end

--========================================================
-- OFFSET
--========================================================

local function getOffset(distance)

    local r = math.random(1, 4)

    if r == 1 then
        return Vector3.new(distance, 0, 0)
    elseif r == 2 then
        return Vector3.new(-distance, 0, 0)
    elseif r == 3 then
        return Vector3.new(0, 0, distance)
    end

    return Vector3.new(0, 0, -distance)

end

local function attack()

    local char = character

    if not char or not char.Parent then
        return
    end

    local tool = char:FindFirstChildOfClass("Tool")

    if tool then

        tool:Activate()

    else

        VirtualUser:CaptureController()

        VirtualUser:Button1Down(
            Vector2.new(500, 500)
        )

        VirtualUser:Button1Up(
            Vector2.new(500, 500)
        )

    end
end

task.spawn(function()

    while true do

        task.wait(LOOP_DELAY)

        if not _G.AutoFarmLevel then

            if lockedTarget then
                clearTarget()
            end

        else

            --================================================
            -- CHARACTER CHECK
            --================================================

            if not character
                or not character.Parent
                or not hrp
                or not hrp.Parent
                or not humanoid
                or not humanoid.Parent
                or humanoid.Health <= 0
            then

                continue

            end

            --================================================
            -- CURRENT TARGET
            --================================================

            local targetHum
            local targetRoot

            if lockedTarget then

                targetHum, targetRoot =
                    getTargetData(lockedTarget)

                if not targetHum or not targetRoot then
                    clearTarget()
                end
            end

            --================================================
            -- SEARCH
            --================================================

            if not lockedTarget then

                local now = os.clock()

                if now - lastSearch >= SEARCH_DELAY then

                    lastSearch = now

                    local target
                    local hum
                    local root

                    target, hum, root =
                        findTarget()

                    if target then

                        lockedTarget = target
                        targetHum = hum
                        targetRoot = root

                        lockedPosition =
                            root.Position
                    end
                end
            end

            --================================================
            -- FARM
            --================================================

            if lockedTarget
                and targetHum
                and targetRoot
                and targetHum.Health > 0
                and lockedPosition
            then

                ----------------------------------------------
                -- LOCK NPC
                ----------------------------------------------

                local pos = targetRoot.Position

                local dx =
                    pos.X - lockedPosition.X

                local dy =
                    pos.Y - lockedPosition.Y

                local dz =
                    pos.Z - lockedPosition.Z

                if dx * dx
                    + dy * dy
                    + dz * dz > 0.01
                then

                    targetRoot.CFrame =
                        CFrame.new(lockedPosition)
                end

                -- Chỉ ghi property khi cần
                if targetHum.WalkSpeed ~= 0 then
                    targetHum.WalkSpeed = 0
                end

                if targetHum.JumpPower ~= 0 then
                    targetHum.JumpPower = 0
                end

                if not targetHum.PlatformStand then
                    targetHum.PlatformStand = true
                end

                ----------------------------------------------
                -- ATTACK
                ----------------------------------------------

                local now = os.clock()

                local speed =
                    tonumber(_G.Settings.Speed)

                if not speed or speed < 0.05 then
                    speed = 0.10
                end

                if now - lastAttack >= speed then

                    lastAttack = now

                    local distance =
                        tonumber(
                            _G.Settings.DashDist
                        ) or 18.2

                    local attackPosition =
                        lockedPosition +
                        getOffset(distance)

                    hrp.CFrame =
                        CFrame.lookAt(
                            attackPosition,
                            lockedPosition
                        )

                    hrp.AssemblyLinearVelocity =
                        Vector3.zero

                    attack()
                end
            end
        end
    end

end)

local function isAreaOccupied(targetPos)
    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player and p.Character and p.Character:FindFirstChild("HumanoidRootPart") then
            if (p.Character.HumanoidRootPart.Position - targetPos).Magnitude <= 7 then
                return true
            end
        end
    end
    for _, v in ipairs(workspace:GetDescendants()) do
        if v:IsA("Seat") or v:IsA("VehicleSeat") then
            if (v.Position - targetPos).Magnitude <= 7 and v.Occupant and v.Occupant.Parent ~= player.Character then 
                return true 
            end
        end
    end
    return false
end

local originalMaterials = {}

local function saveOriginalMaterials()
    if next(originalMaterials) == nil then
        for _, v in ipairs(workspace:GetDescendants()) do
            if v:IsA("BasePart") then
                originalMaterials[v] = {
                    Material = v.Material,
                    Reflectance = v.Reflectance,
                    CastShadow = v.CastShadow
                }
            end
        end
    end
end

local function toggleGraphicReduction(percent, enable)
    saveOriginalMaterials()
    local totalParts = 0
    for _ in pairs(originalMaterials) do totalParts = totalParts + 1 end
    if totalParts == 0 then return end

    local targetCount = math.floor(totalParts * (percent / 100))
    local currentCount = 0

    for part, data in pairs(originalMaterials) do
        if part and part.Parent then
            currentCount = currentCount + 1
            if currentCount <= targetCount then
                if enable then
                    part.Material = Enum.Material.SmoothPlastic
                    part.Reflectance = 0
                    part.CastShadow = false
                else
                    part.Material = data.Material
                    part.Reflectance = data.Reflectance
                    part.CastShadow = data.CastShadow
                end
            end
        end
    end
end

local NPC2_CENTER = Vector3.new(-2790,239,-1762)
local CurrentPoint = 1
local LastTeleport = 0

local POINT_ANGLES = {
    math.rad(-90),
    math.rad(30),
    math.rad(150)
}

local function FindNPC2()
    local npc2 = workspace:FindFirstChild("NPC2",true)
    if not npc2 then return nil,nil end
    if npc2:IsA("Model") then
        local root = npc2:FindFirstChild("HumanoidRootPart") or npc2.PrimaryPart or npc2:FindFirstChildWhichIsA("BasePart")
        if root then return npc2,root end
    elseif npc2:IsA("BasePart") then
        return npc2,npc2
    end
    return nil,nil
end

local function GetCenterPoint(index)
    local distance = tonumber(_G.BossSettings.Mode3.Dist) or 17
    if distance < 0 then distance = 0 end
    local angle = POINT_ANGLES[index] or POINT_ANGLES[1]
    local offset = Vector3.new(math.cos(angle)*distance,0,math.sin(angle)*distance)
    return Vector3.new(NPC2_CENTER.X+offset.X,NPC2_CENTER.Y,NPC2_CENTER.Z+offset.Z)
end

local function TeleportToPoint(index)
    local Character = player.Character
    if not Character then return end
    local HRP = Character:FindFirstChild("HumanoidRootPart")
    if not HRP then return end

    local Boss,BossRoot = FindNPC2()
    if not BossRoot then return end

    local Target = GetCenterPoint(index)
    local LookAt = Vector3.new(BossRoot.Position.X,Target.Y,BossRoot.Position.Z)

    HRP.CFrame = CFrame.lookAt(Target,LookAt)
    HRP.AssemblyLinearVelocity = Vector3.zero
end

local ArenaCenter = Vector3.new(-2791, 239, -1764)
local ArenaRadius = 55
local CameraHeight = 60
local OrbitSpeed = 0.3
local currentAngle = 0
local cameraConnection = nil
local cameraPart = nil

local function SetupArenaCamera(enabled)
    if enabled then
        if cameraConnection then
            cameraConnection:Disconnect()
            cameraConnection = nil
        end
        if cameraPart then
            cameraPart:Destroy()
            cameraPart = nil
        end
        
        cameraPart = Instance.new("Part")
        cameraPart.Name = "ArenaCameraAnchor"
        cameraPart.Size = Vector3.new(1, 1, 1)
        cameraPart.Transparency = 1
        cameraPart.CanCollide = false
        cameraPart.Anchored = true
        cameraPart.Parent = workspace
        
        workspace.CurrentCamera.CameraType = Enum.CameraType.Scriptable
        currentAngle = 0
        
        -- Khởi tạo vị trí ban đầu tại tọa độ cố định
        cameraPart.CFrame = CFrame.new(ArenaCenter)
        
        cameraConnection = RunService.RenderStepped:Connect(function(deltaTime)
            if enabled then
                currentAngle = currentAngle + (OrbitSpeed * deltaTime)
                
                local offsetX = math.cos(currentAngle) * ArenaRadius
                local offsetZ = math.sin(currentAngle) * ArenaRadius
                
                local cameraPos = Vector3.new(
                    ArenaCenter.X + offsetX,
                    ArenaCenter.Y + CameraHeight,
                    ArenaCenter.Z + offsetZ
                )
                
                -- Lerp CFrame camera để chuyển động xoay cực kỳ trơn tru và không bị giật
                local targetCFrame = CFrame.new(cameraPos, ArenaCenter)
                cameraPart.CFrame = cameraPart.CFrame:Lerp(targetCFrame, math.clamp(deltaTime * 20, 0, 1))
                
                workspace.CurrentCamera.CFrame = cameraPart.CFrame
                workspace.CurrentCamera.CameraSubject = cameraPart
            end
        end)
    else
        if cameraConnection then
            cameraConnection:Disconnect()
            cameraConnection = nil
        end
        if cameraPart then
            cameraPart:Destroy()
            cameraPart = nil
        end
        workspace.CurrentCamera.CameraType = Enum.CameraType.Custom
        if Player.Character then
            workspace.CurrentCamera.CameraSubject = Player.Character:FindFirstChildOfClass("Humanoid")
        end
    end
end

local function getClosestPlayer()
    local myRoot = player.Character and player.Character:FindFirstChild("HumanoidRootPart")
    if not myRoot then return nil end

    local closest, shortest = nil, math.huge
    for _, p in pairs(Players:GetPlayers()) do
        if p ~= player then
            local char = p.Character
            local hrp  = char and char:FindFirstChild("HumanoidRootPart")
            local hum  = char and char:FindFirstChildOfClass("Humanoid")
            if hrp and hum and hum.Health > 0 then
                local dist = (myRoot.Position - hrp.Position).Magnitude
                if dist < shortest then
                    shortest = dist
                    closest  = p
                end
            end
        end
    end
    return closest
end

local function EscapeRichText(text)
    text = tostring(text or "")

    text = string.gsub(text, "&", "&amp;")
    text = string.gsub(text, "<", "&lt;")
    text = string.gsub(text, ">", "&gt;")
    text = string.gsub(text, '"', "&quot;")
    text = string.gsub(text, "'", "&apos;")

    return text
end

-- =========================================================
-- COLOR -> HEX
-- =========================================================

local function ColorToHex(color)
    if not color then
        return "#FFFFFF"
    end

    local r = math.floor(color.R * 255 + 0.5)
    local g = math.floor(color.G * 255 + 0.5)
    local b = math.floor(color.B * 255 + 0.5)

    return string.format(
        "#%02X%02X%02X",
        r,
        g,
        b
    )
end

-- =========================================================
-- HEX -> COLOR
-- =========================================================

local function HexToColor(hex)
    hex = tostring(hex or "")

    hex = string.gsub(hex, "#", "")
    hex = string.gsub(hex, "%s+", "")

    if #hex ~= 6 then
        return nil
    end

    local r = tonumber(string.sub(hex, 1, 2), 16)
    local g = tonumber(string.sub(hex, 3, 4), 16)
    local b = tonumber(string.sub(hex, 5, 6), 16)

    if r == nil or g == nil or b == nil then
        return nil
    end

    return Color3.fromRGB(r, g, b)
end

local function NormalizeText(text)
    text = tostring(text or "")

    text = string.lower(text)

    text = string.gsub(
        text,
        "<.->",
        ""
    )

    text = string.gsub(
        text,
        "%s+",
        " "
    )

    text = string.gsub(
        text,
        "^%s*(.-)%s*$",
        "%1"
    )

    return text
end

local function GetGuiCandidates(character)
    local result = {}

    if not character then
        return result
    end

    local exactNames = {
        "RankUI",
        "NameTag",
        "Nametag",
        "NameTagUI",
        "RankGui",
        "RankGUI",
        "Overhead",
        "OverheadGui",
        "PlayerRank",
        "PlayerTag"
    }

    local added = {}

    for _, name in ipairs(exactNames) do
        local object = nil

        pcall(function()
            object = character:FindFirstChild(name, true)
        end)

        if object then
            if object:IsA("BillboardGui")
                or object:IsA("SurfaceGui")
                or object:IsA("Folder")
                or object:IsA("Model")
                or object:IsA("Frame") then

                if not added[object] then
                    added[object] = true
                    table.insert(result, object)
                end
            end
        end
    end

    local descendants = {}

    pcall(function()
        descendants = character:GetDescendants()
    end)

    for _, object in ipairs(descendants) do
        if object:IsA("BillboardGui")
            or object:IsA("SurfaceGui") then

            if not added[object] then
                added[object] = true
                table.insert(result, object)
            end
        end
    end

    return result
end

-- =========================================================
-- FIND RANK UI
-- =========================================================

local function FindValidRankUI()
    local character = LocalPlayer.Character

    if not character then
        return nil
    end

    local candidates = GetGuiCandidates(character)

    local best = nil
    local bestScore = -100000

    for _, gui in ipairs(candidates) do

        if IsLocalCharacterDescendant(gui) then

            local labels = {}

            local descendants = {}

            pcall(function()
                descendants = gui:GetDescendants()
            end)

            if gui:IsA("TextLabel") then
                table.insert(labels, gui)
            end

            for _, object in ipairs(descendants) do
                if object:IsA("TextLabel")
                    and IsLocalCharacterDescendant(object) then

                    table.insert(labels, object)
                end
            end

            if #labels >= 3 then

                local score = #labels * 10

                local guiName =
                    NormalizeText(gui.Name)

                if string.find(guiName, "rank", 1, true) then
                    score = score + 50
                end

                if string.find(guiName, "name", 1, true) then
                    score = score + 35
                end

                if string.find(guiName, "tag", 1, true) then
                    score = score + 30
                end

                if score > bestScore then
                    bestScore = score
                    best = gui
                end
            end
        end
    end

    return best
end

-- =========================================================
-- RESOLVE LABELS
-- =========================================================

local function ResolveLabels(gui)
    local labels = GetLabels(gui)

    if #labels < 3 then
        return false
    end

    -- =====================================================
    -- USERNAME
    -- =====================================================

    local usernameLabel = nil
    local usernameScore = -1

    for _, label in ipairs(labels) do

        local score =
            ScoreUsernameLabel(label)

        if score > usernameScore then
            usernameScore = score
            usernameLabel = label
        end
    end

    -- =====================================================
    -- RANK
    -- =====================================================

    local rankLabel = nil
    local rankScore = -1

    for _, label in ipairs(labels) do

        if label ~= usernameLabel then

            local score =
                ScoreRankLabel(label)

            if score > rankScore then
                rankScore = score
                rankLabel = label
            end
        end
    end

    -- =====================================================
    -- LEVEL
    -- =====================================================

    local levelLabel = nil
    local levelScore = -1

    for _, label in ipairs(labels) do

        if label ~= usernameLabel
            and label ~= rankLabel then

            local score =
                ScoreLevelLabel(label)

            if score > levelScore then
                levelScore = score
                levelLabel = label
            end
        end
    end

    -- =====================================================
    -- FALLBACK
    -- =====================================================

    if not usernameLabel then
        usernameLabel = labels[1]
    end

    if not rankLabel then
        for _, label in ipairs(labels) do
            if label ~= usernameLabel then
                rankLabel = label
                break
            end
        end
    end

    if not levelLabel then
        for _, label in ipairs(labels) do
            if label ~= usernameLabel
                and label ~= rankLabel then

                levelLabel = label
                break
            end
        end
    end

    if not usernameLabel
        or not rankLabel
        or not levelLabel then

        return false
    end

    if not IsLocalCharacterDescendant(usernameLabel)
        or not IsLocalCharacterDescendant(rankLabel)
        or not IsLocalCharacterDescendant(levelLabel) then

        return false
    end

    labelUsername = usernameLabel
    labelDisplay = rankLabel
    labelGroupRank = levelLabel

    return true
end

-- =========================================================
-- UPDATE REFERENCES
-- =========================================================

local function UpdateReferences()
    local character = LocalPlayer.Character

    if not character then
        return false
    end

    if LastCharacter ~= character then

        LastCharacter = character

        RankUIReference = nil

        ClearReferences()

        StableCount = 0
    end

    if RankUIReference then

        if not IsLocalCharacterDescendant(RankUIReference) then

            RankUIReference = nil
            ClearReferences()
            StableCount = 0
        end
    end

    local found = FindValidRankUI()

    if not found then
        StableCount = 0
        UIReady = false
        return false
    end

    if RankUIReference == found then

        StableCount = StableCount + 1

    else

        RankUIReference = found

        StableCount = 1

        ClearReferences()
    end

    if StableCount < StableNeeded then
        return false
    end

    if not ResolveLabels(found) then
        UIReady = false
        return false
    end

    UIReady = true

    return true
end

local function ApplyRainbow(color)
    if not UIReady then
        return
    end

    if not color then
        return
    end

    local level =
        GetLevel()

    -- QUAN TRỌNG:
    -- Ô Name trống => Player
    -- Ô Name có chữ => dùng chữ đó.
    local name =
        GetName()

    -- =====================================================
    -- USERNAME
    -- =====================================================

    if labelUsername
        and IsAlive(labelUsername)
        and IsLocalCharacterDescendant(labelUsername) then

        pcall(function()

            labelUsername.RichText = false
            labelUsername.TextColor3 = color

        end)
    end

    -- =====================================================
    -- RANK
    -- =====================================================

    if labelDisplay
        and IsAlive(labelDisplay)
        and IsLocalCharacterDescendant(labelDisplay) then

        pcall(function()

            labelDisplay.RichText = false
            labelDisplay.TextColor3 = color

        end)
    end

    -- =====================================================
    -- LEVEL + NAME
    -- =====================================================

    if labelGroupRank
        and IsAlive(labelGroupRank)
        and IsLocalCharacterDescendant(labelGroupRank) then

        pcall(function()

            labelGroupRank.RichText = true

            labelGroupRank.Text =
                '<font color="' ..
                ColorToHex(color) ..
                '">(Lv ' ..
                EscapeRichText(level) ..
                ') ' ..
                EscapeRichText(name) ..
                '</font>'

        end)
    end
end

-- =========================================================
-- UPDATE LABELS
-- =========================================================

local function UpdateLabels()
    if not UIReady then
        return
    end

    local character =
        LocalPlayer.Character

    if not character then
        UIReady = false
        return
    end

    if LastCharacter ~= character then
        UIReady = false
        return
    end

    if not RankUIReference
        or not IsLocalCharacterDescendant(RankUIReference) then

        UIReady = false
        RankUIReference = nil

        return
    end

    if not labelUsername
        or not labelDisplay
        or not labelGroupRank then

        UIReady = false
        return
    end

    if not IsLocalCharacterDescendant(labelUsername)
        or not IsLocalCharacterDescendant(labelDisplay)
        or not IsLocalCharacterDescendant(labelGroupRank) then

        UIReady = false
        ClearReferences()

        return
    end

    if RainbowEnabled then
        ApplyRainbow(GetRainbowColor())
    else
        ApplyNormal()
    end
end

-- =========================================================
-- STOP PICKER
-- =========================================================

local function StopPicker()
    if PickerConnection then

        pcall(function()
            PickerConnection:Disconnect()
        end)

        PickerConnection = nil
    end
end

-- =========================================================
-- UPDATE SV
-- =========================================================

local function UpdateSV(x, y)

    x = math.clamp(x, 0, 1)
    y = math.clamp(y, 0, 1)

    CurrentSaturation = x
    CurrentValue = 1 - y

    if SVMarker then
        SVMarker.Position =
            UDim2.new(
                x,
                0,
                y,
                0
            )
    end

    SetTargetColor(
        Color3.fromHSV(
            CurrentHue,
            CurrentSaturation,
            CurrentValue
        )
    )
end

-- =========================================================
-- UPDATE HUE
-- =========================================================

local function UpdateHue(y)

    y = math.clamp(y, 0, 1)

    CurrentHue = 1 - y

    if HueMarker then
        HueMarker.Position =
            UDim2.new(
                0,
                -3,
                y,
                -3
            )
    end

    SetTargetColor(
        Color3.fromHSV(
            CurrentHue,
            CurrentSaturation,
            CurrentValue
        )
    )
end

-- =========================================================
-- PICKER DRAG
-- =========================================================

local function ConnectPickerDrag(frame, callback)

    frame.InputBegan:Connect(function(input)

        local valid =
            input.UserInputType
            == Enum.UserInputType.MouseButton1
            or input.UserInputType
            == Enum.UserInputType.Touch

        if not valid then
            return
        end

        local function UpdateInput(inputObject)

            if not ColorPicker
                or not ColorPicker.Visible then
                return
            end

            local position =
                frame.AbsolutePosition

            local size =
                frame.AbsoluteSize

            if size.X <= 0
                or size.Y <= 0 then
                return
            end

            callback(
                inputObject,
                position,
                size
            )
        end

        UpdateInput(input)

        StopPicker()

        PickerConnection =
            UserInputService.InputChanged:Connect(
                function(move)

                    local validMove =
                        move.UserInputType
                        == Enum.UserInputType.MouseMovement
                        or move.UserInputType
                        == Enum.UserInputType.Touch

                    if validMove then
                        UpdateInput(move)
                    end

                end
            )
    end)
end

local function OpenColorPicker(target)

    if not ColorPicker then
        return
    end

    StopPicker()

    ColorTarget = target

    local color =
        GetTargetColor(target)

    local h, s, v =
        Color3.toHSV(color)

    CurrentHue = h
    CurrentSaturation = s
    CurrentValue = v

    if ColorPickerTitle then
        ColorPickerTitle.Text =
            "MÀU " ..
            string.upper(target)
    end

    if ColorPreview then
        ColorPreview.BackgroundColor3 =
            color
    end

    if HexBox then
        HexBox.Text =
            ColorToHex(color)
    end

    if SVMarker then
        SVMarker.Position =
            UDim2.new(
                s,
                0,
                1 - v,
                0
            )
    end

    if HueMarker then
        HueMarker.Position =
            UDim2.new(
                0,
                -3,
                1 - h,
                -3
            )
    end

    ColorPicker.Visible = true
end

-- =========================================================
-- DRAG
-- =========================================================

local function MakeDraggable(frame, dragObject)

    dragObject.InputBegan:Connect(
        function(input)

            local valid =
                input.UserInputType
                == Enum.UserInputType.MouseButton1
                or input.UserInputType
                == Enum.UserInputType.Touch

            if not valid then
                return
            end

            local dragging = true

            local dragStart =
                input.Position

            local startPosition =
                frame.Position

            if MainDragConnection then

                pcall(function()
                    MainDragConnection:Disconnect()
                end)

                MainDragConnection = nil
            end

            MainDragConnection =
                UserInputService.InputChanged:Connect(
                    function(move)

                        if not dragging then
                            return
                        end

                        local validMove =
                            move.UserInputType
                            == Enum.UserInputType.MouseMovement
                            or move.UserInputType
                            == Enum.UserInputType.Touch

                        if not validMove then
                            return
                        end

                        local delta =
                            move.Position
                            - dragStart

                        frame.Position =
                            UDim2.new(
                                startPosition.X.Scale,
                                startPosition.X.Offset + delta.X,
                                startPosition.Y.Scale,
                                startPosition.Y.Offset + delta.Y
                            )

                    end
                )

            input.Changed:Connect(
                function()

                    if input.UserInputState
                        == Enum.UserInputState.End then

                        dragging = false

                        if MainDragConnection then

                            pcall(function()
                                MainDragConnection:Disconnect()
                            end)

                            MainDragConnection = nil
                        end
                    end

                end
            )

        end
    )
end

local function ApplyFromMenu()

    if Applying then
        return
    end

    Applying = true

    -- USERNAME
    if BoxUser then

        local value =
            tostring(BoxUser.Text or "")

        if value ~= "" then
            customUsername = value
        end
    end

    -- RANK
    if BoxRank then

        local value =
            tostring(BoxRank.Text or "")

        if value ~= "" then
            customDisplay = value
        end
    end

    -- FIND UI
    if not UIReady then
        UpdateReferences()
    end

    if not UIReady then

        SetStatus(
            "Không tìm thấy Rank UI",
            Color3.fromRGB(
                255,
                150,
                80
            )
        )

        Applying = false
        return
    end

    -- APPLY
    if RainbowEnabled then
        ApplyRainbow(
            GetRainbowColor()
        )
    else
        ApplyNormal()
    end

    SetStatus(
        "Đã áp dụng",
        Color3.fromRGB(
            80,
            255,
            150
        )
    )

    if ApplyBtn then

        ApplyBtn.Text =
            "ĐÃ ÁP DỤNG"

        task.delay(
            0.8,
            function()

                if ApplyBtn then
                    ApplyBtn.Text =
                        "ÁP DỤNG"
                end

                if StatusBar then
                    SetStatus(
                        "Sẵn sàng",
                        Color3.fromRGB(
                            180,
                            130,
                            255
                        )
                    )
                end

            end
        )
    end

    Applying = false
end

LocalPlayer.CharacterAdded:Connect(
    function(character)

        LastCharacter = character

        RankUIReference = nil

        ClearReferences()

        StableCount = 0

        task.spawn(
            function()

                task.wait(1)

                local found = false

                for i = 1, 35 do

                    if LocalPlayer.Character
                        ~= character then

                        return
                    end

                    if UpdateReferences() then

                        found = true
                        break
                    end

                    task.wait(0.35)
                end

                if found and UIReady then

                    if RainbowEnabled then
                        ApplyRainbow(
                            GetRainbowColor()
                        )
                    else
                        ApplyNormal()
                    end
                end

            end
        )
    end
)

-- =========================================================
-- CHARACTER REMOVING
-- =========================================================

LocalPlayer.CharacterRemoving:Connect(
    function(character)

        if LastCharacter == character then

            UIReady = false

            RankUIReference = nil

            ClearReferences()

            StableCount = 0
        end

    end
)

-- =========================================================
-- AUTO RECONNECT
-- =========================================================

task.spawn(
    function()

        while true do

            task.wait(0.75)

            local character =
                LocalPlayer.Character

            if character then

                if LastCharacter ~= character then

                    LastCharacter = character

                    RankUIReference = nil

                    ClearReferences()

                    StableCount = 0

                end

                if RankUIReference then

                    if not IsLocalCharacterDescendant(
                        RankUIReference
                    ) then

                        RankUIReference = nil

                        ClearReferences()

                        StableCount = 0
                    end
                end

                if not UIReady then
                    UpdateReferences()
                end

            else

                UIReady = false
            end
        end

    end
)

task.defer(function()
    while not Fluent.Options or not Fluent.Options["UseConfigSystemToggle"] or not Fluent.Options["SelectConfigDropdown"] do
        task.wait(0.2)
    end
    
    if isfile(GlobalSettingsFile) then
        pcall(function()
            local data = HttpService:JSONDecode(readfile(GlobalSettingsFile))
            if data and data["SelectedConfigName"] then
                _G.SelectedConfigName = data["SelectedConfigName"]
                _G.UseConfigSystem = data["UseConfigSystem"] or false
                
                Fluent.Options["UseConfigSystemToggle"]:SetValue(_G.UseConfigSystem)
                Fluent.Options["SelectConfigDropdown"]:SetValue(_G.SelectedConfigName)
                
                if _G.UseConfigSystem then
                    loadSelectedConfig(_G.SelectedConfigName)
                    Fluent:Notify({Title = "VND💵 System", Content = "Đã tự động nạp & kích hoạt: " .. _G.SelectedConfigName, Duration = 5})
                end
            end
        end)
    end
end)
